\section{Complete 15-Dimensional Cancer Model System}

The complete cancer dynamics are governed by the following system of differential equations:

\begin{align}
\frac{dN_1}{dt} &=  - \frac{I_{1} N_{1} \beta_{1}}{0.01 N_{1} + 0.01 N_{2} + 0.01 Q + 0.01 R_{1} + 0.01 R_{2} + 0.01 S + 1} - N_{1} \kappa_{Q} \left(0.5 H + 1\right) - N_{1} \kappa_{S} \left(1.3 - 0.3 G\right) \left(\eta_{C} u_{C} + \eta_{E} u_{E} + \eta_{H} u_{H} + \eta_{I} u_{I}\right) + \frac{N_{1} \lambda_{1} \left(1 - \frac{N_{1} + N_{2} + Q + R_{1} + R_{2} + S}{K}\right) \left(0.1 M + 1\right)}{M acidosis_{factor} + 1} - N_{1} \omega_{R1} \left(2 - G\right) \left(\eta_{C} u_{C} + \eta_{E} u_{E} + \eta_{H} u_{H} + \eta_{I} u_{I}\right) - N_{1} \omega_{R2} \left(2 - G\right) \left(\eta_{C} u_{C} + \eta_{E} u_{E} + \eta_{H} u_{H} + \eta_{I} u_{I}\right) - N_{1} \left(\eta_{C} u_{C} + \eta_{E} u_{E} + \eta_{H} u_{H} + \eta_{I} u_{I}\right) \\
\frac{dN_2}{dt} &=  - \frac{0.5 I_{1} N_{2} \beta_{1}}{0.01 N_{1} + 0.01 N_{2} + 0.01 Q + 0.01 R_{1} + 0.01 R_{2} + 0.01 S + 1} - N_{2} \kappa_{Q} \left(0.5 H + 1\right) + \frac{N_{2} \lambda_{2} \left(1 - \frac{N_{1} + N_{2} + Q + R_{1} + R_{2} + S}{K}\right) \left(0.1 M + 1\right)}{M acidosis_{factor} + 1} - N_{2} \left(0.7 \eta_{C} u_{C} + 0.7 \eta_{E} u_{E} + 0.7 \eta_{H} u_{H} + 0.7 \eta_{I} u_{I}\right) \\
\frac{dI_1}{dt} &=  - \frac{I_{1} I_{2} \beta_{2}}{I_{1} + 1} - I_{1} \delta_{I} \left(0.2 H + 1\right) + 0.1 I_{1} \eta_{I} u_{I} + \phi_{1} + \frac{\phi_{2} \left(N_{1} + N_{2} + Q + R_{1} + R_{2} + S\right)}{0.01 N_{1} + 0.01 N_{2} + 0.01 Q + 0.01 R_{1} + 0.01 R_{2} + 0.01 S + 1} \\
\frac{dI_2}{dt} &=  - I_{2} \delta_{I} \left(0.1 H + 1\right) - 0.1 I_{2} \eta_{I} u_{I} + \frac{\phi_{3} \left(N_{1} + N_{2} + Q + R_{1} + R_{2} + S\right)}{0.01 N_{1} + 0.01 N_{2} + 0.01 Q + 0.01 R_{1} + 0.01 R_{2} + 0.01 S + 1} \\
\frac{dP}{dt} &=  - P \delta_{P} + \gamma \left(0.5 H + 1\right) \left(0.3 M + 1\right) \left(N_{1} + N_{2} + Q + R_{1} + R_{2} + S\right) \\
\frac{dA}{dt} &=  - A \delta_{A} + \frac{\alpha_{A} \left(H + 1\right) \left(N_{1} + N_{2} + Q + R_{1} + R_{2} + S\right)}{0.01 N_{1} + 0.01 N_{2} + 0.01 Q + 0.01 R_{1} + 0.01 R_{2} + 0.01 S + 1} \\
\frac{dQ}{dt} &=  - \frac{Q \lambda_{Q} \left(0.2 A + 1\right)}{0.5 H + 1} + \kappa_{Q} \left(0.5 H + 1\right) \left(N_{1} + N_{2}\right) \\
\frac{dR_1}{dt} &=  - I_{1} R_{1} \beta_{1} \rho_{1} + N_{1} \omega_{R1} \left(2 - G\right) \left(\eta_{C} u_{C} + \eta_{E} u_{E} + \eta_{H} u_{H} + \eta_{I} u_{I}\right) + R_{1} lambdaR_{1} \left(1 - \frac{N_{1} + N_{2} + Q + R_{1} + R_{2} + S}{K}\right) \\
\frac{dR_2}{dt} &=  - I_{1} R_{2} \beta_{1} \rho_{2} + N_{1} \omega_{R2} \left(2 - G\right) \left(\eta_{C} u_{C} + \eta_{E} u_{E} + \eta_{H} u_{H} + \eta_{I} u_{I}\right) + R_{2} lambdaR_{2} \left(1 - \frac{N_{1} + N_{2} + Q + R_{1} + R_{2} + S}{K}\right) \\
\frac{dS}{dt} &=  N_{1} \kappa_{S} \left(1.3 - 0.3 G\right) \left(\eta_{C} u_{C} + \eta_{E} u_{E} + \eta_{H} u_{H} + \eta_{I} u_{I}\right) - S \delta_{S} \\
\frac{dD}{dt} &=  - D k_{el} + u_{C} + u_{E} + u_{H} + u_{I} \\
\frac{dD_m}{dt} &=  D k_{el} - Dm clearance \\
\frac{dG}{dt} &=  - G \mu \left(0.5 H + \eta_{C} u_{C} + \eta_{E} u_{E} + \eta_{H} u_{H} + \eta_{I} u_{I} + 1\right) - 0.001 G + 0.001 \\
\frac{dM}{dt} &=  H M metabolic_{switch rate} - 0.05 M \\
\frac{dH}{dt} &=  - 0.1 A H + 0.1 \max\left(0, \frac{- hypoxia_{threshold} + \frac{N_{1} + N_{2} + Q + R_{1} + R_{2} + S}{K}}{1 - hypoxia_{threshold}}\right)
\end{align}
