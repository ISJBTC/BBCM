df3/dt = -I1*I2*beta2/(I1 + 1) - I1*delta_I*(0.2*H + 1) + 0.1*I1*eta_I*u_I + phi1 + phi2*(N1 + N2 + Q + R1 + R2 + S)/(0.01*N1 + 0.01*N2 + 0.01*Q + 0.01*R1 + 0.01*R2 + 0.01*S + 1)

df5/dt = -P*delta_P + gamma*(0.5*H + 1)*(0.3*M + 1)*(N1 + N2 + Q + R1 + R2 + S)

df11/dt = -D*k_el + u_C + u_E + u_H + u_I

df14/dt = H*M*metabolic_switch_rate - 0.05*M

df15/dt = -0.1*A*H + 0.1*Max(0, (-hypoxia_threshold + (N1 + N2 + Q + R1 + R2 + S)/K)/(1 - hypoxia_threshold))
